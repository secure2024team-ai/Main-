local function track(connection)
    Connections[#Connections + 1] = connection
    return connection
end

local function create(className, props, parent)
    local obj = Instance.new(className)
    if props then
        for key, value in pairs(props) do
            obj[key] = value
        end
    end
    obj.Parent = parent
    return obj
end

local function corner(parent, radius)
    return create("UICorner", {CornerRadius = UDim.new(0, radius)}, parent)
end

local function cornerFull(parent)
    return create("UICorner", {CornerRadius = UDim.new(0.5, 0)}, parent)
end

local function stroke(parent, color, thickness, transparency)
    return create("UIStroke", {
        Color = color or C.Border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
    }, parent)
end

local function gradient(parent, c0, c1, rotation)
    return create("UIGradient", {
        Color = ColorSequence.new(c0, c1),
        Rotation = rotation or 90,
    }, parent)
end

local function shade(parent)
    return gradient(parent, Color3.new(1, 1, 1), Color3.fromRGB(205, 200, 222), 90)
end

local function label(parent, text, size, color, weight, extra)
    local l = create("TextLabel", {
        BackgroundTransparency = 1,
        Text = text,
        TextSize = size,
        TextColor3 = color or C.Text,
        FontFace = F[weight or "Medium"],
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center,
        TextTruncate = Enum.TextTruncate.AtEnd,
        Size = UDim2.new(1, 0, 0, size + 6),
    }, nil)
    if extra then
        for key, value in pairs(extra) do
            l[key] = value
        end
    end
    l.Parent = parent
    return l
end

-- Theme binding: fn(accentColor) -> value for property. Updates in place on theme change.
local function bind(inst, prop, fn)
    local entry = {inst = inst, prop = prop, fn = fn or function(a) return a end}
    Bound[#Bound + 1] = entry
    inst[prop] = entry.fn(Accent)
    return entry
end

local function applyBound()
    for _, entry in ipairs(Bound) do
        if entry.inst.Parent then
            pcall(function()
                entry.inst[entry.prop] = entry.fn(Accent)
            end)
        end
    end
end

local function tween(obj, duration, props, style, direction)
    if Dead then return nil end
    if Settings.ReducedMotion then
        for key, value in pairs(props) do
            obj[key] = value
        end
        return nil
    end
    local t = TweenService:Create(obj, TweenInfo.new(
        duration,
        style or Enum.EasingStyle.Quad,
        direction or Enum.EasingDirection.Out
    ), props)
    t:Play()
    return t
end

local function val(x)
    if type(x) == "function" then
        return x()
    end
    return x
end

local useHover = UserInputService.MouseEnabled

local function pressable(btn, rest, hover, pressed)
    local inside, down = false, false
    local function refresh()
        if btn:GetAttribute("Disabled") then return end
        local color
        if down then
            color = val(pressed)
        elseif inside and useHover then
            color = val(hover)
        else
            color = val(rest)
        end
        tween(btn, 0.12, {BackgroundColor3 = color})
    end
    track(btn.MouseEnter:Connect(function() inside = true refresh() end))
    track(btn.MouseLeave:Connect(function() inside = false down = false refresh() end))
    track(btn.MouseButton1Down:Connect(function() down = true refresh() end))
    track(btn.MouseButton1Up:Connect(function() down = false refresh() end))
end

local function clampSafe(v, lo, hi)
    return math.max(lo, math.min(hi, v))
end

-- Keeps `target` (anchor 0.5) on screen. keepX/keepY = pixels that must stay visible.
local function placeClamped(gui_, target, startPos, dx, dy, keepX, keepY)
    local V = gui_.AbsoluteSize
    local size = target.AbsoluteSize
    local cx = V.X * startPos.X.Scale + startPos.X.Offset + dx
    local cy = V.Y * startPos.Y.Scale + startPos.Y.Offset + dy
    cx = clampSafe(cx, keepX - size.X / 2, V.X - keepX + size.X / 2)
    cy = clampSafe(cy, size.Y / 2, V.Y - keepY + size.Y / 2)
    target.Position = UDim2.new(
        startPos.X.Scale, cx - V.X * startPos.X.Scale,
        startPos.Y.Scale, cy - V.Y * startPos.Y.Scale
    )
end

-- Single drag dispatcher shared by window, floating button and sliders.
local function bindDrag(handle, onBegin, onMove, onEnd)
    track(handle.InputBegan:Connect(function(input)
        if ActiveDrag and ActiveDrag.input.UserInputState ~= Enum.UserInputState.End then
            return
        end
        local t = input.UserInputType
        if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
            ActiveDrag = {input = input, kind = t, onMove = onMove, onEnd = onEnd}
            onBegin(input)
        end
    end))
end

track(UserInputService.InputChanged:Connect(function(input)
    local d = ActiveDrag
    if not d then return end
    if (d.kind == Enum.UserInputType.Touch and input == d.input)
        or (d.kind == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement) then
        d.onMove(input)
    end
end))

track(UserInputService.InputEnded:Connect(function(input)
    local d = ActiveDrag
    if not d then return end
    if (d.kind == Enum.UserInputType.Touch and input == d.input)
        or (d.kind == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseButton1) then
        ActiveDrag = nil
        d.onEnd(input)
    end
end))

local function resolveIcons()
    local id = GENX_ICON_ID
    if type(id) ~= "string" or id == "" or string.find(id, "YOUR_IMAGE_ID", 1, true) then
        return -- placeholder: keep the "G" fallback
    end
    for _, icon in ipairs(IconRegistry) do
        icon.image.Image = id
        icon.image.Visible = true
    end
    task.spawn(function()
        local deadline = os.clock() + 8
        while not Dead and os.clock() < deadline do
            local all = true
            for _, icon in ipairs(IconRegistry) do
                if icon.image.IsLoaded then
                    icon.fallback.Visible = false
                else
                    all = false
                end
            end
            if all then return end
            task.wait(0.25)
        end
        if Dead then return end
        for _, icon in ipairs(IconRegistry) do
            if not icon.image.IsLoaded then
                icon.image.Visible = false
                icon.fallback.Visible = true
            end
        end
    end)
end

-- 6. TOASTS (Single notification, 2 seconds duration)
----------------------------------------------------------------
local toastCounter = 0
toast = function(text, kind, force)
    if Dead or (not Settings.Toasts and not force) then return end
    toastCounter += 1

    -- Clear any active toast immediately so only one shows up at a time
    for _, child in ipairs(toastHost:GetChildren()) do
        if child:IsA("CanvasGroup") then
            child:Destroy()
        end
    end

    local color = Accent
    if kind == "warn" then color = C.Amber elseif kind == "error" then color = C.Red end

    local t = create("CanvasGroup", {
        LayoutOrder = toastCounter,
        Size = UDim2.new(1, 0, 0, 38),
        BackgroundColor3 = C.Card,
        BorderSizePixel = 0,
        GroupTransparency = 1,
    }, toastHost)
    corner(t, 12)
    stroke(t, color, 1, 0.35)
    local dot = create("Frame", {
        Position = UDim2.fromOffset(13, 15),
        Size = UDim2.fromOffset(8, 8),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
    }, t)

updateAccentLoop = function()
    local shouldRun = (not Dead) and isOpen and Settings.AnimatedAccents and not Settings.ReducedMotion
    if shouldRun then
        if not lineTween then
            lineGradient.Offset = Vector2.new(-0.7, 0)
            lineTween = TweenService:Create(lineGradient, TweenInfo.new(
                2.6, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true
            ), {Offset = Vector2.new(0.7, 0)})
            lineTween:Play()
        end
    elseif lineTween then
        lineTween:Cancel()
        lineTween = nil
        lineGradient.Offset = Vector2.new(0, 0)
    end
end

setOpen = function(open)
    if Dead then return end
    open = open == true
    if open == isOpen and window.Visible == open then return end
    isOpen = open
    animToken += 1
    local token = animToken

    if open then
        window.Visible = true
        if Settings.ReducedMotion then
            body.Position = UDim2.new()
            body.GroupTransparency = 0
        else
            body.Position = UDim2.fromOffset(0, 16)
            body.GroupTransparency = 1
            tween(body, 0.3, {Position = UDim2.new(), GroupTransparency = 0}, Enum.EasingStyle.Quint)
        end
        fadeBackdrop(true)
        if refreshLive then refreshLive() end
        if refreshMovement then refreshMovement() end
    else
        tween(body, 0.18, {Position = UDim2.fromOffset(0, 10), GroupTransparency = 1})
        fadeBackdrop(false)
        local delay = Settings.ReducedMotion and 0 or 0.2
        task.delay(delay, function()
            if not Dead and animToken == token and not isOpen then
                window.Visible = false
            end
        end)
        if not Settings.FloatingButton and not UserInputService.KeyboardEnabled then
            toast("Tap the reopen button to bring GenX back", "warn", true)
        end
    end

    updateFloating()
    updateAccentLoop()
end

local swatchRings = {}

-- Extra theme styling (gradient colors + tinted backgrounds).
-- Themes without color2/bg/surface (Purple and the plain ones) get the
-- original look back exactly.
local function applyThemeExtras(theme)
    if theme.color2 then
        glowGradient.Color = ColorSequence.new(theme.color, theme.color2)
        lineGradient.Color = ColorSequence.new(theme.color, theme.color2)
    else
        local plain = ColorSequence.new(Color3.new(1, 1, 1))
        glowGradient.Color = plain
        lineGradient.Color = plain
    end
    tween(body, 0.3, {BackgroundColor3 = theme.bg or C.Body})
    tween(sidebar, 0.3, {BackgroundColor3 = theme.surface or C.Surface})
    tween(content, 0.3, {BackgroundColor3 = theme.surface or C.Surface})
end

setAccent = function(name, silent)
    local theme = ThemeByName[name]
    if not theme then return end
    AccentName = name
    CurrentTheme = theme
    applyThemeExtras(theme)
    for themeName, ring in pairs(swatchRings) do
        ring.Transparency = (themeName == name) and 0 or 1
    end
    if accentNameLabel then
        accentNameLabel.Text = string.upper(name)
    end
    if accentTween then
        accentTween:Cancel()
        accentTween = nil
    end
    if Settings.ReducedMotion then
        accentValue.Value = theme.color
        applyBound()
    else
        accentTween = TweenService:Create(accentValue, TweenInfo.new(
            0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out
        ), {Value = theme.color})
        accentTween:Play()
    end
    if not silent then
        toast("Accent set to " .. name)
    end
    emit("Accent", name)
end

local function showPage(name, instant)
    pageToken += 1
    for pageName, p in pairs(PageList) do
        if pageName ~= name then
            if p.tween then p.tween:Cancel() p.tween = nil end
            p.group.Visible = false
            p.group.GroupTransparency = 1
        end
    end
    local p = PageList[name]
    if not p then return end
    if p.tween then p.tween:Cancel() p.tween = nil end
    p.scroll.CanvasPosition = Vector2.zero
    p.group.Visible = true
    if instant or Settings.ReducedMotion then
        p.group.Position = UDim2.new()
        p.group.GroupTransparency = 0
    else
        p.group.Position = UDim2.fromOffset(0, 12)
        p.group.GroupTransparency = 1
        p.tween = tween(p.group, 0.24, {Position = UDim2.new(), GroupTransparency = 0}, Enum.EasingStyle.Quint)
    end
end

local function tabGoals(item)
    if item.selected then
        return Accent:Lerp(C.Surface, 0.82), 0, Accent, C.Text
    end
    return C.Card, 1, C.Muted, C.Muted
end

local function refreshTab(item, animate)
    local bg, bgT, iconColor, titleColor = tabGoals(item)
    if animate then
        tween(item.button, 0.2, {BackgroundColor3 = bg, BackgroundTransparency = bgT})
        tween(item.icon, 0.2, {TextColor3 = iconColor})
        if item.paint then item.paint(iconColor, true) end
        tween(item.title, 0.2, {TextColor3 = titleColor})
    else
        item.button.BackgroundColor3 = bg
        item.button.BackgroundTransparency = bgT
        item.icon.TextColor3 = iconColor
        if item.paint then item.paint(iconColor, false) end
        item.title.TextColor3 = titleColor
    end
end

local function positionIndicator(animate)
    local item = TabByName[ActiveTab]
    if not item then return end
    local pos, size
    if Layout.compact then
        local tabW = Layout.NavW / #TabItems
        size = UDim2.fromOffset(22, 3)
        pos = UDim2.fromOffset(math.floor((item.index - 0.5) * tabW - 11), Layout.NavH - 4)
    else
        size = UDim2.fromOffset(3, math.max(14, Layout.tabH - 18))
        pos = UDim2.fromOffset(3, (item.index - 1) * (Layout.tabH + 4) + 9)
    end
    if animate then
        tween(indicator, 0.28, {Position = pos, Size = size}, Enum.EasingStyle.Quart)
    else
        indicator.Position = pos
        indicator.Size = size
    end
end

local function layoutTabs()
    local n = #TabItems
    for i, item in ipairs(TabItems) do
        if Layout.compact then
            item.button.Position = UDim2.new((i - 1) / n, 0, 0, 0)
            item.button.Size = UDim2.new(1 / n, 0, 1, 0)
            item.icon.Position = UDim2.fromOffset(0, 3)
            item.icon.Size = UDim2.new(1, 0, 0, 24)
            item.icon.TextXAlignment = Enum.TextXAlignment.Center
            item.title.Position = UDim2.fromOffset(0, 28)
            item.title.Size = UDim2.new(1, 0, 0, 14)
            item.title.TextSize = 10
            item.title.TextXAlignment = Enum.TextXAlignment.Center
            if item.visual then item.visual.Position = UDim2.new(0.5, 0, 0, 15) end
        else
            item.button.Position = UDim2.fromOffset(0, (i - 1) * (Layout.tabH + 4))
            item.button.Size = UDim2.new(1, 0, 0, Layout.tabH)
            item.icon.Position = UDim2.fromOffset(10, 0)
            item.icon.Size = UDim2.fromOffset(26, Layout.tabH)
            item.icon.TextXAlignment = Enum.TextXAlignment.Center
            item.title.Position = UDim2.fromOffset(44, 0)
            item.title.Size = UDim2.new(1, -50, 1, 0)
            item.title.TextSize = 13
            item.title.TextXAlignment = Enum.TextXAlignment.Left
            if item.visual then item.visual.Position = UDim2.new(0, 23, 0.5, 0) end
        end
    end
end

local function selectTab(name, instant)
    if ActiveTab == name then return end
    ActiveTab = name
    for _, item in ipairs(TabItems) do
        item.selected = item.name == name
        refreshTab(item, not instant)
    end
    positionIndicator(not instant)
    showPage(name, instant)
    if refreshMovement then refreshMovement() end
end

-- Built-in vector icons (22x22), drawn from frames so they recolor with the theme
local FULL = UDim.new(0.5, 0)
local function buildGlyph(name, parent)
    local g = create("Frame", {
        Name = "Glyph",
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(0.5, 0.5),
        Size = UDim2.fromOffset(22, 22),
    }, parent)
    local parts = {}

    local function box(x, y, w, h, rot, radius)
        local f = create("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(w, h),
            Rotation = rot or 0,
            BorderSizePixel = 0,
            BackgroundColor3 = C.Muted,
        }, g)
        if radius then create("UICorner", {CornerRadius = radius}, f) end
        parts[#parts + 1] = {f, "BackgroundColor3"}
    end

    local function ring(x, y, w, h, thick, radius)
        local f = create("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset(x, y),
            Size = UDim2.fromOffset(w, h),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
        }, g)
        create("UICorner", {CornerRadius = radius}, f)
        local st = create("UIStroke", {Thickness = thick, Color = C.Muted}, f)
        parts[#parts + 1] = {st, "Color"}
    end

    if name == "Home" then
        box(7, 7.5, 11, 2.2, -35, UDim.new(0, 1))      -- roof left
        box(15, 7.5, 11, 2.2, 35, UDim.new(0, 1))      -- roof right
        ring(11, 14.5, 11, 8, 2, UDim.new(0, 2))       -- walls
        box(11, 16.5, 3, 5, 0, UDim.new(0, 1))         -- door
    elseif name == "Visuals" then
        ring(11, 11, 20, 11, 2, FULL)                  -- eye
        box(11, 11, 6, 6, 0, FULL)                     -- pupil
    elseif name == "Combat" then
        ring(11, 11, 13, 13, 2, FULL)                  -- crosshair ring
        box(11, 11, 3, 3, 0, FULL)                     -- center dot
        box(11, 2.5, 2, 5, 0)                          -- ticks
        box(11, 19.5, 2, 5, 0)
        box(2.5, 11, 5, 2, 0)
        box(19.5, 11, 5, 2, 0)
    elseif name == "Movement" then
        box(11.5, 10.5, 18, 2.4, -45, UDim.new(0, 1))  -- arrow shaft
        box(13.5, 5.5, 9, 2.4, 0, UDim.new(0, 1))      -- arrow head
        box(17, 10, 2.4, 9, 0, UDim.new(0, 1))
    elseif name == "Settings" then
        ring(11, 11, 11, 11, 2.5, FULL)                -- gear body
        for k = 0, 7 do
            local a = math.rad(k * 45)
            box(11 + 8.6 * math.cos(a), 11 + 8.6 * math.sin(a), 3.4, 3.4, k * 45, UDim.new(0, 1))
        end
    else
        box(11, 11, 8, 8, 0, FULL)
    end

    local function paint(color, animate)
        for _, p in ipairs(parts) do
            if animate then
                tween(p[1], 0.2, {[p[2]] = color})
            else
                p[1][p[2]] = color
            end
        end
    end
    return g, paint
end

        end
        icon.TextTransparency = 1
        task.spawn(function()
            local deadline = os.clock() + 8
            while not Dead and os.clock() < deadline and not img.IsLoaded do task.wait(0.25) end
            if not Dead and not img.IsLoaded then
                img.Visible = false
                icon.TextTransparency = 0
            end
        end)

    bind(icon, "TextColor3", function(a) local _, _, c = tabGoals(item) return c end)

        -- re-tint on theme change (the fn paints as a side effect)
        bind(item.visual, "Name", function(a)

            item.paint(c, false)

        end)

    track(btn.MouseButton1Down:Connect(function()
        if not item.selected then

        end
    end))
    track(btn.MouseButton1Up:Connect(function()
        if not item.selected then
            tween(btn, 0.15, {BackgroundTransparency = 1})
        end
    end))
    track(btn.MouseLeave:Connect(function()
        if not item.selected then
            tween(btn, 0.15, {BackgroundTransparency = 1})
        end
    end))
    track(btn.Activated:Connect(function()
        selectTab(def.name, false)
    end))
end

local function statGrid(page)

        CellPadding = UDim2.fromOffset(10, 10),
        SortOrder = Enum.SortOrder.LayoutOrder,
    }, grid)
    -- Height comes from the card count (2 per row), so it can't over-measure
    local function fit()
        local n = 0
        for _, c in ipairs(grid:GetChildren()) do
            if c:IsA("Frame") then n += 1 end
        end
        local rows = math.ceil(n / 2)
        grid.Size = UDim2.new(1, 0, 0, rows > 0 and (rows * 84 + (rows - 1) * 10) or 0)
    end
    grid.ChildAdded:Connect(fit)
    return grid
end

        AutoButtonColor = false,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 14, 0, 36),

    }, card)
    local rail = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.new(1, 0, 0, 6),
        BackgroundColor3 = C.Border,
        BorderSizePixel = 0,
    }, hit)
    cornerFull(rail)
    local fill = create("Frame", {
        Size = UDim2.fromScale(0, 1),
        BorderSizePixel = 0,
    }, rail)
    cornerFull(fill)
    bind(fill, "BackgroundColor3")
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.fromScale(0, 0.5),
        Size = UDim2.fromOffset(18, 18),
        BackgroundColor3 = C.Text,
        BorderSizePixel = 0,
        ZIndex = 2,
    }, hit)
    cornerFull(knob)
    local knobStroke = stroke(knob, Accent, 2, 0)
    bind(knobStroke, "Color")

    local function render()
        local r = (value - opts.min) / (opts.max - opts.min)
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, 0, 0.5, 0)
        valueLabel.Text = opts.format and opts.format(value) or tostring(value)
    end

    local function set(v, silent)
        v = math.clamp(v, opts.min, opts.max)
        local step = opts.step or 1
        v = opts.min + math.floor((v - opts.min) / step + 0.5) * step
        v = math.clamp(v, opts.min, opts.max)
        if v == value then return end
        value = v
        Settings[opts.id] = v
        render()
        if opts.onChange then opts.onChange(v, silent == true) end
        emit(opts.id, v)
    end

    local function fromInput(input)
        local w = hit.AbsoluteSize.X
        if w <= 0 then return end
        local r = math.clamp((input.Position.X - hit.AbsolutePosition.X) / w, 0, 1)
        set(opts.min + r * (opts.max - opts.min), false)
    end

    bindDrag(hit,
        function(input)
            page.scroll.ScrollingEnabled = false
            tween(knob, 0.1, {Size = UDim2.fromOffset(22, 22)})
            fromInput(input)
        end,
        fromInput,
        function()
            page.scroll.ScrollingEnabled = true
            tween(knob, 0.12, {Size = UDim2.fromOffset(18, 18)})
            if opts.toastOnEnd then
                toast(opts.title .. "  ·  " .. (opts.format and opts.format(value) or tostring(value)))
            end
        end
    )

    render()
    Controls[opts.id] = {Set = set, Default = value}
    return {Set = set, Get = function() return value end}
end

local function makeButton(parent, text, style, order, callback)
    style = style or "secondary"
    local btn = create("TextButton", {
        Name = "Button",
        LayoutOrder = order or 0,
        Size = UDim2.new(1, 0, 0, 42),
        BackgroundColor3 = C.Card,
        BorderSizePixel = 0,
        AutoButtonColor = false,
        Text = text,
        TextSize = 13,
        TextColor3 = C.Text,
        FontFace = F.Semi,
    }, parent)
    corner(btn, 11)
    local st = stroke(btn, C.Border, 1, 0.1)
    local obj = {Button = btn, Enabled = true}

    local function rest()
        if style == "primary" then return Accent:Lerp(Color3.new(0, 0, 0), 0.38) end
        if style == "danger" then return Color3.fromRGB(58, 22, 34) end
        return C.Card
    end
    local function hover()
        if style == "primary" then return Accent:Lerp(Color3.new(0, 0, 0), 0.25) end
        if style == "danger" then return Color3.fromRGB(78, 29, 45) end
        return C.CardHi
    end
    local function pressedColor()
        if style == "primary" then return Accent:Lerp(Color3.new(0, 0, 0), 0.55) end
        if style == "danger" then return Color3.fromRGB(40, 16, 25) end
        return C.CardPress
    end

    if style == "primary" then
        bind(btn, "BackgroundColor3", rest)
        bind(st, "Color")
    elseif style == "danger" then
        btn.TextColor3 = C.Red
        st.Color = Color3.fromRGB(120, 45, 62)
    end
    btn.BackgroundColor3 = rest()
    pressable(btn, rest, hover, pressedColor)

    function obj.SetEnabled(enabled)
        obj.Enabled = enabled
        btn:SetAttribute("Disabled", not enabled)
        tween(btn, 0.15, {
            BackgroundColor3 = enabled and rest() or C.Surface,
            TextTransparency = enabled and 0 or 0.6,
        })
        st.Transparency = enabled and 0.1 or 0.6
    end
    function obj.SetText(newText)
        btn.Text = newText
    end

    track(btn.Activated:Connect(function()
        if obj.Enabled then playClick() end
        if obj.Enabled and callback then callback() end
    end))
    return obj
end

-- Writes the current UI state into the loaded aimlock config
local function aimApply()
    local cfg = aimCfg()
    if not cfg then return end
    for _, item in ipairs(combatToggles) do
        cfg[item.key] = Settings[item.id] == true
    end
    cfg.FOVRadius = Settings.Aim_FOVRadius
    cfg.MaxDistance = Settings.Aim_MaxDistance
    cfg.Smoothing = Settings.Aim_Smoothing
end

-- Loads the aimlock once (called at startup and on first toggle). After loading,
-- everything is forced OFF, then the UI state is applied, so only what the
-- user toggled is active.
local aimLoaded, aimLoading = false, false
local function aimEnsure()
    if aimLoaded then
        aimApply()
        return
    end
    if aimLoading then return end
    aimLoading = true
    task.spawn(function()
        local ok, err = pcall(function()
            -- ?t= busts GitHub's raw cache so the newest file is always fetched
            loadstring(game:HttpGet("https://raw.githubusercontent.com/marvens7999/Esp/refs/heads/main/Aimlock?t=" .. tostring(os.time())))()
        end)
        aimLoading = false
        if Dead then return end
        if not ok then
            warn("[GenX] Aimlock load error: " .. tostring(err))
            toast("Aimlock failed to load", "error")
            return
        end
        local cfg = aimCfg()
        if not cfg then
            -- Old/stale Aimlock file (no shared config): kill it so it can't lock on by itself
            pcall(function()
                local env = getgenv()
                if env.DeltaAimlockUnload then env.DeltaAimlockUnload() end
            end)
            warn("[GenX] Aimlock file on GitHub is outdated (no Aimlock_Config). Unloaded it.")
            toast("Aimlock file outdated - update GitHub", "error", true)
            return
        end
        aimLoaded = true
        for _, item in ipairs(combatToggles) do cfg[item.key] = false end
        aimApply()
    end)
end

local CurrentHumanoid = nil
local charConns = {}

local function clearCharConns()
    for _, c in ipairs(charConns) do
        pcall(function() c:Disconnect() end)
    end
    table.clear(charConns)
end

local function setCharStatus(text, color)
    charCard.Value.Text = text
    charCard.Value.TextColor3 = color
end

local function refreshHealth()
    if Dead then return end
    local hum = CurrentHumanoid
    if not hum or not hum.Parent then
        healthText.Text = "--"
        tween(healthFill, 0.2, {Size = UDim2.fromScale(0, 1)})
        return
    end
    local maxHealth = math.max(hum.MaxHealth, 1)
    local hp = math.clamp(hum.Health, 0, maxHealth)
    local pct = hp / maxHealth
    healthText.Text = string.format("%d / %d", math.floor(hp + 0.5), math.floor(maxHealth + 0.5))
    local color = pct > 0.6 and C.Green or (pct > 0.3 and C.Amber or C.Red)
    tween(healthFill, 0.2, {Size = UDim2.fromScale(pct, 1), BackgroundColor3 = color})
    if hp <= 0 then
        setCharStatus("DOWN", C.Red)
    else
        setCharStatus("ALIVE", C.Green)
    end
end

local function bindCharacter(char)
    clearCharConns()
    CurrentHumanoid = nil
    if not char then
        setCharStatus("NO CHARACTER", C.Muted)
        refreshHealth()
        return
    end
    setCharStatus("LOADING", C.Muted)
    task.spawn(function()
        local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 10)
        if Dead or not hum or LocalPlayer.Character ~= char then
            if not Dead and LocalPlayer.Character == char then
                setCharStatus("UNAVAILABLE", C.Amber)
            end
            return
        end
        CurrentHumanoid = hum
        charConns[#charConns + 1] = hum.HealthChanged:Connect(refreshHealth)
        charConns[#charConns + 1] = hum:GetPropertyChangedSignal("MaxHealth"):Connect(refreshHealth)
        charConns[#charConns + 1] = hum.Died:Connect(function()
            setCharStatus("RESPAWNING", C.Amber)
            refreshHealth()
        end)
        refreshHealth()
        if refreshMovement then refreshMovement() end
    end)
end

track(LocalPlayer.CharacterAdded:Connect(bindCharacter))
track(LocalPlayer.CharacterRemoving:Connect(function()
    clearCharConns()
    CurrentHumanoid = nil
    setCharStatus("RESPAWNING", C.Amber)
    refreshHealth()
end))
bindCharacter(LocalPlayer.Character)

refreshMovement = function()
    if Dead then return end
    Move.gravity.Value.Text = string.format("%.1f", workspace.Gravity)
    local hum = CurrentHumanoid
    if hum and hum.Parent then
        Move.walk.Value.Text = string.format("%.1f", hum.WalkSpeed)
        Move.walk.Sub.Text = "studs / sec"
        if hum.UseJumpPower then
            Move.jump.Value.Text = string.format("%.0f", hum.JumpPower)
            Move.jump.Sub.Text = "Jump power"
        else
            Move.jump.Value.Text = string.format("%.1f", hum.JumpHeight)
            Move.jump.Sub.Text = "Jump height"
        end
        Move.state.Value.Text = hum:GetState().Name
    else
        Move.walk.Value.Text = "--"
        Move.walk.Sub.Text = "No character"
        Move.jump.Value.Text = "--"
        Move.jump.Sub.Text = "No character"
        Move.state.Value.Text = "--"
    end
end

-- 16B. MOVEMENT FEATURES (runtime, driven by the Movement tab switches)
----------------------------------------------------------------
do
    local RunService = game:GetService("RunService")
    local S = Settings
    local st = {
        hum = nil, stateConn = nil,
        origSpeed = nil, origJump = nil,
        noclip = {},
        fly = false,
        floating = false, floatY = 0, armed = true, jumped = false,
        jumpDown = false, jumpStart = 0, held = false, taps = {},
        safePos = nil, safeT = 0,
        bounds = nil, boundsT = 0, boundsBusy = false,
        noGroundT = nil, tpCool = 0,
        controls = nil,
    }

    Mv.parts = function()
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if hum and root and hum.Health > 0 then return char, hum, root end
        return nil
    end

    -- Joystick vector (x = right, z = back) in camera space
    Mv.getMove = function(hum, cam)
        if not st.controls then
            pcall(function()
                local pm = LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule", 1)
                st.controls = require(pm):GetControls()
            end)
        end
        if st.controls then
            local ok, v = pcall(function() return st.controls:GetMoveVector() end)
            if ok and v then return v end
        end
        return cam.CFrame:VectorToObjectSpace(hum.MoveDirection)
    end

    -- Puts everything back the way it was
    Mv.restore = function()
        pcall(function() RunService:UnbindFromRenderStep("GenXMvInput") end)
        local _, hum = Mv.parts()
        if hum then
            if st.origSpeed then hum.WalkSpeed = st.origSpeed end
            if st.origJump then
                if hum.UseJumpPower then hum.JumpPower = st.origJump else hum.JumpHeight = st.origJump end
            end
            if st.fly then hum.PlatformStand = false end
        end
        for part in pairs(st.noclip) do
            if part.Parent then part.CanCollide = true end
        end
        table.clear(st.noclip)
        if st.stateConn then pcall(function() st.stateConn:Disconnect() end) st.stateConn = nil end
        st.origSpeed, st.origJump, st.fly, st.floating = nil, nil, false, false
    end

    -- Reads the jump button every frame: tap counting + hold detection
    RunService:BindToRenderStep("GenXMvInput", Enum.RenderPriority.Last.Value, function()
        if Dead then return end
        local _, hum = Mv.parts()
        local down = hum ~= nil and hum.Jump == true
        local now = os.clock()
        if down and not st.jumpDown then
            st.jumpDown = true
            st.jumpStart = now
            st.taps[#st.taps + 1] = now
        elseif not down then
            st.jumpDown = false
        end
        st.held = down and (now - st.jumpStart) > 0.25
        for i = #st.taps, 1, -1 do
            if now - st.taps[i] > 0.9 then table.remove(st.taps, i) end
        end
    end)

    -- Noclip (runs before physics)
    track(RunService.Stepped:Connect(function()
        local char = LocalPlayer.Character
        if S.Mv_Noclip and char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                    st.noclip[part] = true
                end
            end
        elseif next(st.noclip) then
            for part in pairs(st.noclip) do
                if part.Parent then part.CanCollide = true end
            end
            table.clear(st.noclip)
        end
    end))

    -- Infinite Jump
    track(UserInputService.JumpRequest:Connect(function()
        if not S.Mv_InfJump then return end
        local _, hum = Mv.parts()
        if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
    end))

    -- Tap Teleport
    local function tapTeleport(pos)
        if not S.Mv_TapTP then return end
        local char, _, root = Mv.parts()
        local cam = workspace.CurrentCamera
        if not (root and cam) then return end
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {char}
        local ray = cam:ViewportPointToRay(pos.X, pos.Y)
        local hit = workspace:Raycast(ray.Origin, ray.Direction * 3000, params)
        if hit then
            root.CFrame = CFrame.new(hit.Position + Vector3.new(0, 4, 0))
            root.AssemblyLinearVelocity = Vector3.zero
        end
    end
    track(UserInputService.TouchTapInWorld:Connect(function(pos, processed)
        if not processed then tapTeleport(pos) end
    end))
    track(UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            tapTeleport(UserInputService:GetMouseLocation())
        end
    end))

    -- Map bounds scan (cached, refreshed in the background)
    local function scanBounds()
        if st.boundsBusy then return end
        st.boundsBusy = true
        task.spawn(function()
            local minV, maxV
            local n = 0
            for _, d in ipairs(workspace:GetDescendants()) do
                n += 1
                if n % 400 == 0 then task.wait() end
                if d:IsA("BasePart") and d.CanCollide and d.Size.Magnitude > 4
                    and d.Size.Magnitude < 3000
                    and not (d.Parent and d.Parent:FindFirstChildOfClass("Humanoid")) then
                    local p, h = d.Position, d.Size / 2
                    local lo, hi = p - h, p + h
                    minV = minV and Vector3.new(math.min(minV.X, lo.X), math.min(minV.Y, lo.Y), math.min(minV.Z, lo.Z)) or lo
                    maxV = maxV and Vector3.new(math.max(maxV.X, hi.X), math.max(maxV.Y, hi.Y), math.max(maxV.Z, hi.Z)) or hi
                end
            end
            if minV then st.bounds = {min = minV, max = maxV} end
            st.boundsT = os.clock()
            st.boundsBusy = false
        end)
    end

    -- Main per-frame logic
    track(RunService.Heartbeat:Connect(function(dt)
        local _, hum, root = Mv.parts()
        if not hum then return end
        local now = os.clock()

        -- new character: reset state
        if hum ~= st.hum then
            st.hum = hum
            st.origSpeed, st.origJump = nil, nil
            st.fly, st.floating, st.armed, st.jumped = false, false, true, false
            if st.stateConn then pcall(function() st.stateConn:Disconnect() end) end
            st.stateConn = hum.StateChanged:Connect(function(_, new)
                if new == Enum.HumanoidStateType.Jumping and st.armed then st.jumped = true end
            end)
        end

        local grounded = hum.FloorMaterial ~= Enum.Material.Air

        -- WalkSpeed
        if S.Mv_Speed then
            if st.origSpeed == nil then st.origSpeed = hum.WalkSpeed end
            hum.WalkSpeed = S.Mv_SpeedVal
        elseif st.origSpeed then
            hum.WalkSpeed = st.origSpeed
            st.origSpeed = nil
        end

        -- JumpPower
        if S.Mv_Jump then
            if st.origJump == nil then
                st.origJump = hum.UseJumpPower and hum.JumpPower or hum.JumpHeight
            end
            if hum.UseJumpPower then
                hum.JumpPower = S.Mv_JumpVal
            else
                hum.JumpHeight = (S.Mv_JumpVal * S.Mv_JumpVal) / (2 * workspace.Gravity)
            end
        elseif st.origJump then
            if hum.UseJumpPower then hum.JumpPower = st.origJump else hum.JumpHeight = st.origJump end
            st.origJump = nil
        end

        -- Auto Jump
        if S.Mv_AutoJump and grounded and not st.fly then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end

        -- Fly
        if S.Mv_Fly then
            if not st.fly then
                st.fly = true
                st.floating = false
                hum.PlatformStand = true
            end
            local cam = workspace.CurrentCamera
            if cam then
                local mv = Mv.getMove(hum, cam)
                local look, right = cam.CFrame.LookVector, cam.CFrame.RightVector
                local dir = look * (-mv.Z) + right * mv.X
                if st.held then dir += Vector3.new(0, 1, 0) end
                if dir.Magnitude > 1 then dir = dir.Unit end
                root.AssemblyLinearVelocity = dir * S.Mv_FlySpeed
                local flat = Vector3.new(look.X, 0, look.Z)
                if flat.Magnitude > 0.01 then
                    root.CFrame = CFrame.lookAt(root.Position, root.Position + flat)
                end
            end
        elseif st.fly then
            st.fly = false
            hum.PlatformStand = false
            root.AssemblyLinearVelocity = Vector3.zero
        end

        -- Auto Float: jump = stay at that height, hold jump = rise, tap jump 3x = come down
        if S.Mv_AutoFloat and not st.fly then
            if grounded then
                st.armed = true
                st.floating = false
            elseif st.armed and not st.floating and st.jumped
                and root.AssemblyLinearVelocity.Y <= 0 then
                st.floating = true
                st.floatY = root.Position.Y
                st.armed = false
                st.jumped = false
                table.clear(st.taps)
            end
            if st.floating then
                if #st.taps >= 3 then
                    st.floating = false
                    table.clear(st.taps)
                else
                    if st.held then st.floatY += 28 * dt end
                    local v = root.AssemblyLinearVelocity
                    local vy = math.clamp((st.floatY - root.Position.Y) * 10, -50, 60)
                    root.AssemblyLinearVelocity = Vector3.new(v.X, vy, v.Z)
                end
            end
        else
            st.floating = false
        end

        -- Anti-Void (fast): bounds check + ground raycast + fall detection
        local char = LocalPlayer.Character
        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Exclude
        rp.FilterDescendantsInstances = {char}
        local groundHit = workspace:Raycast(root.Position, Vector3.new(0, -80, 0), rp)

        -- record safe spot: standing on real ground, not mid-fly
        if grounded and groundHit and not st.fly and now - st.safeT > 0.25 then
            st.safePos = root.Position
            st.safeT = now
        end

        if S.Mv_AntiVoid then
            if not st.bounds or now - st.boundsT > 15 then scanBounds() end

            local pos = root.Position
            local off = false

            -- 1) outside the whole map's bounding box
            local b = st.bounds
            if b then
                local m = 60
                if pos.Y < b.min.Y - 25
                    or pos.X < b.min.X - m or pos.X > b.max.X + m
                    or pos.Z < b.min.Z - m or pos.Z > b.max.Z + m then
                    off = true
                end
            end

            -- 2) falling with nothing below for a split second
            if not groundHit and root.AssemblyLinearVelocity.Y < -15 and not st.fly then
                st.noGroundT = st.noGroundT or now
                if now - st.noGroundT > 0.12 then off = true end
            else
                st.noGroundT = nil
            end

            -- 3) engine void line
            if pos.Y < workspace.FallenPartsDestroyHeight + 80 then off = true end

            if off and st.safePos and now - st.tpCool > 0.3 then
                st.tpCool = now
                st.noGroundT = nil
                root.AssemblyLinearVelocity = Vector3.zero
                root.AssemblyAngularVelocity = Vector3.zero
                root.CFrame = CFrame.new(st.safePos + Vector3.new(0, 4, 0))
                st.floating = false
            end
        else
            st.noGroundT = nil
        end
    end))
end

    local ring = stroke(swatch, C.Text, 2, theme.name == AccentName and 0 or 1)
    swatchRings[theme.name] = ring
    track(swatch.Activated:Connect(function()
        setAccent(theme.name, false)
    end))
end

layout = function()
    if Dead then return end
    local V = gui.AbsoluteSize
    if V.X < 50 or V.Y < 50 then return end

    local landscape = V.X > V.Y
    local marginX = (landscape and UserInputService.TouchEnabled) and 28 or 12
    local marginY = 14

    local base = math.clamp(math.min(V.X / 380, V.Y / 380), 0.78, 1)
    local target = 1 - math.clamp(AUTO_SCALE_TARGET, 0, 60) / 100
    local s = base * target * (Settings.CompactMode and 0.9 or 1)
    s = math.clamp(s, 0.45, 1.2)

    local availW = (V.X - marginX * 2) / s
    local availH = (V.Y - marginY * 2) / s
    local W = math.floor(math.max(280, math.min(availW, 760)))
    local H = math.floor(math.max(300, math.min(availH, 520)))
    local compact = W < 560

    Layout.W, Layout.H, Layout.compact = W, H, compact

    window.Size = UDim2.fromOffset(math.floor(W * s), math.floor(H * s))
    body.Size = UDim2.fromOffset(W, H)
    bodyScale.Scale = s

    local r = math.floor(18 * s + 0.5)
    borderCorner.CornerRadius = UDim.new(0, r)
    shadowCorner.CornerRadius = UDim.new(0, r)
    haloCorner.CornerRadius = UDim.new(0, r)

    if compact then
        local navH = 58
        sidebar.Position = UDim2.fromOffset(8, H - navH - 8)
        sidebar.Size = UDim2.fromOffset(W - 16, navH)
        content.Position = UDim2.fromOffset(8, HEADER_H + 7)
        content.Size = UDim2.fromOffset(W - 16, H - HEADER_H - 7 - navH - 16)
        sidebarCaption.Visible = false
        nav.Position = UDim2.fromOffset(4, 4)
        nav.Size = UDim2.new(1, -8, 1, -8)
        Layout.NavW = W - 16 - 8
        Layout.NavH = navH - 8
        footerLeft.Visible = false
        footerRight.Visible = false
    else
        local sideW = 152
        local top = HEADER_H + 8
        local h = H - top - FOOTER_H - 6
        sidebar.Position = UDim2.fromOffset(10, top)
        sidebar.Size = UDim2.fromOffset(sideW, h)
        content.Position = UDim2.fromOffset(10 + sideW + 8, top)
        content.Size = UDim2.fromOffset(W - (10 + sideW + 8) - 10, h)
        sidebarCaption.Visible = true
        nav.Position = UDim2.fromOffset(6, 34)
        nav.Size = UDim2.new(1, -12, 1, -40)
        Layout.NavW = sideW - 12
        Layout.NavH = h - 40
        Layout.tabH = math.clamp(math.floor((Layout.NavH - (#TabItems - 1) * 4) / #TabItems), 34, 120)
        footerLeft.Visible = true
        footerRight.Visible = true
    end

    layoutTabs()
    positionIndicator(false)

    toastHost.Size = UDim2.fromOffset(math.min(320, V.X - marginX * 2), 0)
    toastHost.Position = UDim2.new(0.5, 0, 0, marginY + 6)

    placeClamped(gui, window, window.Position, 0, 0, 90, 48)
    local fsize = floatRoot.AbsoluteSize
    placeClamped(gui, floatRoot, floatRoot.Position, 0, 0, fsize.X, fsize.Y)
end

track(gui:GetPropertyChangedSignal("AbsoluteSize"):Connect(layout))

local function cleanup()
    if Dead then return end
    Dead = true

    if _G.__GenXCleanup == cleanup then
        _G.__GenXCleanup = nil
    end

    if accentTween then pcall(function() accentTween:Cancel() end) end
    if lineTween then pcall(function() lineTween:Cancel() end) end
    pcall(function() accentValue:Destroy() end)

    for _, connection in ipairs(Connections) do
        pcall(function() connection:Disconnect() end)
    end
    table.clear(Connections)
    clearCharConns()

    if Mv.restore then pcall(Mv.restore) end
    table.clear(Bound)
    table.clear(Controls)
    table.clear(TouchRows)
    ActiveDrag = nil
end

track(gui.Destroying:Connect(cleanup))
_G.__GenXCleanup = function()
    if gui and gui.Parent then
        gui:Destroy()
    else
        cleanup()
    end
end
